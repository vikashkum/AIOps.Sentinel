  # View backend logs
  kubectl logs -n {{ include "aiops-sentinel.namespace" . }} -l app.kubernetes.io/component=backend -f

  # Describe a pod (troubleshooting)
  kubectl describe pod -n {{ include "aiops-sentinel.namespace" . }} -l app.kubernetes.io/component=backend

  # Open a shell inside the backend container
  kubectl exec -it -n {{ include "aiops-sentinel.namespace" . }} \
    $(kubectl get pod -n {{ include "aiops-sentinel.namespace" . }} \
    -l app.kubernetes.io/component=backend -o name | head -1) -- /bin/bash

  # View events (helpful when pods are stuck)
  kubectl get events -n {{ include "aiops-sentinel.namespace" . }} --sort-by='.lastTimestamp'

  # Scale backend replicas manually
  kubectl scale deployment {{ include "aiops-sentinel.backend.name" . }} \
    -n {{ include "aiops-sentinel.namespace" . }} --replicas=2
